eoe_wre_settlement.0002 = {
	type = character_event
	hidden = yes
	trigger = {
		has_variable = eoe_wre_settlement_scheduled
		OR = { has_title = title:e_western_roman_empire has_title = title:h_western_roman_empire }
	}
	immediate = {
		if = {
			limit = { any_realm_county = { holder = root has_variable = eoe_wre_pending_settlement } }
			save_scope_as = eoe_wre_settlement_emperor
			random_realm_county = {
				limit = { holder = scope:eoe_wre_settlement_emperor has_variable = eoe_wre_pending_settlement }
				save_scope_as = eoe_wre_package_anchor
				de_jure_liege = { save_scope_as = eoe_wre_package_duchy }
				if = {
					limit = { NOT = { has_variable = eoe_wre_package_refusals } }
					set_variable = { name = eoe_wre_package_refusals value = 0 }
				}
			}
			if = {
				limit = { has_character_flag = eoe_wre_local_settlement_policy }
				scope:eoe_wre_package_anchor.var:eoe_wre_previous_holder ?= { save_scope_as = eoe_wre_settlement_candidate }
				if = {
					limit = {
						exists = scope:eoe_wre_settlement_candidate
						scope:eoe_wre_settlement_candidate = { eoe_valid_wre_local_restoration_candidate = yes }
					}
					scope:eoe_wre_settlement_candidate = {
						add_character_flag = eoe_wre_settlement_offer_pending
						trigger_event = { id = eoe_wre_settlement.0010 days = 1 }
					}
				}
				else = {
					remove_character_flag = eoe_wre_settlement_candidate_found
					if = {
						limit = {
							exists = scope:eoe_wre_settlement_candidate
							exists = scope:eoe_wre_settlement_candidate.dynasty
						}
						scope:eoe_wre_settlement_candidate.dynasty = {
							every_dynasty_member = {
								limit = { eoe_valid_wre_local_family_candidate = yes }
								add_to_list = eoe_wre_local_family_candidates
							}
						}
						if = {
							limit = { any_in_list = { list = eoe_wre_local_family_candidates always = yes } }
							random_in_list = {
								list = eoe_wre_local_family_candidates
								weight = { base = 1 }
								scope:eoe_wre_settlement_emperor = { add_character_flag = eoe_wre_settlement_candidate_found }
								add_character_flag = eoe_wre_local_family_offer
								add_character_flag = eoe_wre_settlement_offer_pending
								trigger_event = { id = eoe_wre_settlement.0010 days = 1 }
							}
						}
					}
					if = {
						limit = { NOT = { has_character_flag = eoe_wre_settlement_candidate_found } }
						eoe_create_and_grant_wre_local_package_effect = yes
						trigger_event = { id = eoe_wre_settlement.0002 days = 1 }
					}
					remove_character_flag = eoe_wre_settlement_candidate_found
				}
			}
			else = {
				if = {
					limit = { scope:eoe_wre_package_anchor.var:eoe_wre_package_refusals >= 2 }
					eoe_create_and_grant_wre_imperial_package_effect = yes
					trigger_event = { id = eoe_wre_settlement.0002 days = 1 }
				}
				else = {
					remove_character_flag = eoe_wre_settlement_candidate_found
					every_courtier_or_guest = {
						limit = {
							eoe_valid_wre_imperial_settlement_candidate = yes
							exists = dynasty
							OR = { dynasty = scope:eoe_wre_settlement_emperor.dynasty prestige_level >= 3 }
						}
						add_to_list = eoe_wre_imperial_settlement_candidates
					}
					every_vassal = {
						limit = { is_landed = yes }
						every_courtier_or_guest = {
							limit = {
								eoe_valid_wre_local_family_candidate = yes
								eoe_valid_wre_imperial_settlement_candidate = yes
								exists = dynasty
								OR = { dynasty = scope:eoe_wre_settlement_emperor.dynasty prestige_level >= 3 }
							}
							add_to_list = eoe_wre_imperial_settlement_candidates
						}
					}
					if = {
						limit = { any_in_list = { list = eoe_wre_imperial_settlement_candidates always = yes } }
						random_in_list = {
							list = eoe_wre_imperial_settlement_candidates
							weight = { base = 1 }
							scope:eoe_wre_settlement_emperor = { add_character_flag = eoe_wre_settlement_candidate_found }
							add_character_flag = eoe_wre_settlement_offer_pending
							trigger_event = { id = eoe_wre_settlement.0011 days = 1 }
						}
					}
					if = {
						limit = { NOT = { has_character_flag = eoe_wre_settlement_candidate_found } }
						eoe_create_and_grant_wre_imperial_package_effect = yes
						trigger_event = { id = eoe_wre_settlement.0002 days = 1 }
					}
					remove_character_flag = eoe_wre_settlement_candidate_found
				}
			}
		}
		else = {
			if = {
				limit = { any_realm_county = { has_variable = eoe_wre_pending_provincial_settlement } }
				save_scope_as = eoe_wre_settlement_emperor
				random_realm_county = {
					limit = { has_variable = eoe_wre_pending_provincial_settlement }
					save_scope_as = eoe_wre_provincial_anchor
					de_jure_liege = { save_scope_as = eoe_wre_package_duchy }
					if = {
						limit = { NOT = { has_variable = eoe_wre_provincial_refusals } }
						set_variable = { name = eoe_wre_provincial_refusals value = 0 }
					}
				}
				if = {
					limit = {
						scope:eoe_wre_package_duchy = { eoe_wre_provincial_title_available = yes }
						scope:eoe_wre_provincial_anchor.var:eoe_wre_provincial_refusals < 2
					}
					every_realm_county = {
						limit = {
							has_variable = eoe_wre_pending_provincial_settlement
							de_jure_liege = scope:eoe_wre_package_duchy
							holder = {
								eoe_valid_wre_provincial_governor_candidate = yes
								OR = {
									scope:eoe_wre_settlement_emperor = { has_character_flag = eoe_wre_local_settlement_policy }
									AND = {
										scope:eoe_wre_settlement_emperor = { has_character_flag = eoe_wre_imperial_settlement_policy }
										culture = scope:eoe_wre_settlement_emperor.culture
										faith = scope:eoe_wre_settlement_emperor.faith
										OR = {
											dynasty = scope:eoe_wre_settlement_emperor.dynasty
											prestige_level >= 3
										}
									}
								}
							}
						}
						holder = { add_to_list = eoe_wre_provincial_candidates }
					}
					if = {
						limit = { any_in_list = { list = eoe_wre_provincial_candidates always = yes } }
						random_in_list = {
							list = eoe_wre_provincial_candidates
							weight = {
								base = 10
								modifier = { add = 20 culture = scope:eoe_wre_provincial_anchor.title_province.culture }
								modifier = { add = 15 prestige_level >= 3 }
								modifier = { add = stewardship }
								modifier = {
									add = 25
									scope:eoe_wre_settlement_emperor = { has_character_flag = eoe_wre_imperial_settlement_policy }
									dynasty = scope:eoe_wre_settlement_emperor.dynasty
								}
							}
							save_scope_as = eoe_wre_provincial_governor
							add_character_flag = eoe_wre_provincial_offer_pending
							trigger_event = { id = eoe_wre_settlement.0012 days = 1 }
						}
					}
					else = {
						eoe_clear_wre_provincial_package_effect = yes
						trigger_event = { id = eoe_wre_settlement.0002 days = 1 }
					}
				}
				else = {
					eoe_clear_wre_provincial_package_effect = yes
					trigger_event = { id = eoe_wre_settlement.0002 days = 1 }
				}
			}
			else = {
				remove_variable = eoe_wre_settlement_scheduled
				remove_character_flag = eoe_wre_imperial_settlement_policy
				remove_character_flag = eoe_wre_local_settlement_policy
				trigger_event = eoe_wre_settlement.0020
			}
		}
	}
}

eoe_wre_settlement.0010 = {
	type = character_event
	title = eoe_wre_settlement.0010.t
	desc = eoe_wre_settlement.0010.desc
	theme = realm
	override_background = { reference = throne_room }
	left_portrait = { character = root animation = thinking }
	right_portrait = { character = scope:eoe_wre_settlement_emperor animation = scepter }
	trigger = {
		has_character_flag = eoe_wre_settlement_offer_pending
		exists = scope:eoe_wre_settlement_emperor
		exists = scope:eoe_wre_package_duchy
		scope:eoe_wre_package_anchor = { has_variable = eoe_wre_pending_settlement }
	}
	option = {
		name = eoe_wre_settlement.0010.a
		remove_character_flag = eoe_wre_settlement_offer_pending
		if = {
			limit = { has_character_flag = eoe_wre_local_family_offer }
			remove_character_flag = eoe_wre_local_family_offer
			eoe_grant_wre_duchy_package_effect = yes
		}
		else = { eoe_grant_wre_previous_holder_package_effect = yes }
		scope:eoe_wre_settlement_emperor = { trigger_event = { id = eoe_wre_settlement.0002 days = 1 } }
		ai_chance = {
			base = 70
			modifier = { add = 30 has_trait = ambitious }
			modifier = { add = -40 has_trait = zealous faith != scope:eoe_wre_settlement_emperor.faith }
			modifier = { add = -50 has_relation_rival = scope:eoe_wre_settlement_emperor }
		}
	}
	option = {
		name = eoe_wre_settlement.0010.b
		remove_character_flag = eoe_wre_settlement_offer_pending
		remove_character_flag = eoe_wre_local_family_offer
		add_character_flag = { flag = eoe_wre_settlement_refused years = 10 }
		scope:eoe_wre_settlement_emperor = {
			eoe_create_and_grant_wre_local_package_effect = yes
			trigger_event = { id = eoe_wre_settlement.0002 days = 1 }
		}
		ai_chance = { base = 30 }
	}
}
