# Culture scope
is_roman = {
	custom_description = {
		text = eoe_is_roman_requirement_tt
		OR = {
			culture = culture:roman
			AND = {
				culture = {
					any_parent_culture_or_above = {
						custom_tooltip = {
							text = parent_roman
							this = culture:roman
						}
					}
				}
				OR = {
					culture = { has_cultural_pillar = heritage_latin }
					culture = { has_cultural_tradition = tradition_sundered_soul }
					culture = { has_cultural_tradition = tradition_imperium }
					culture = { has_cultural_tradition = tradition_sacrum_imperium }
				}
				NOR = {
					culture = culture:sardinian
					culture = {
						any_parent_culture_or_above = {
							custom_tooltip = {
								text = parent_sardinian
								this = culture:sardinian
							}
						}
					}
				}
			}
		}
	}
}


is_greek_or_roman = {
	custom_description = {
		text = eoe_is_greek_or_roman_requirement_tt
		OR = {
			is_roman = yes
			culture = { has_cultural_pillar = heritage_byzantine }
			culture = { has_cultural_pillar = heritage_ancient_greek }
		}
	}
}
