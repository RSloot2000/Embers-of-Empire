eoe_roman_event.0001 = {
        type = character_event
        title = eoe_roman_events.0001.title
        desc = eoe_roman_events.0001.desc
        theme = byzantine
    override_background = { reference = throne_room }
    override_effect_2d = { reference = legend_glow }

    left_portrait = {
        character = root
        animation = war_over_win
    }
    immediate = {
        every_culture_global = {              
            if = {      
                limit = {
                    AND = {
                        has_cultural_pillar = language_latin
                        has_cultural_pillar = heritage_latin
                    }
                }
            remove_culture_tradition = tradition_sundered_soul
            }
        }
    }
    option = { 
        trigger = {
		    custom_tooltip = {
			    text = must_be_hellenic
                OR = {
                    has_religion = religion:hellenism_religion
                    faith = {
                        has_doctrine = tenet_hellenic_syncretism
                    }
                }
            }
        }  
        name = eoe_roman_event.0001.a
        culture = {
            add_culture_tradition = tradition_imperium
        }
        hidden_effect = { # Latins' chance to convert.
			every_vassal_or_below = {
				limit = {
					capital_province = {
						geographical_region = world_europe_south_italy
					}
					is_ai = yes
				}
				random = {
					chance = 60
					set_culture_same_as = root
					capital_county = {
						set_county_culture = root.culture
                        set_county_faith = root.faith
					}
				}
			}
		}
    }
    option = {
        trigger = {
		    custom_tooltip = {
			    text = must_be_christian
                has_religion = religion:christianity_religion
            }
        } 
        name = eoe_roman_event.0001.b
        culture = {
            add_culture_tradition = tradition_sacrum_imperium
        }
    }
    after = {
        every_culture_global = {              
            if = {      
                limit = {
                    AND = {
                        has_cultural_pillar = language_latin
                        has_cultural_pillar = heritage_latin
                    }
                }
                add_culture_tradition = tradition_molded_architecture
            }
        }
    }
}

eoe_roman_event.0004 = {
        type = character_event
        title = eoe_roman_event.0004.title
        desc = eoe_roman_event.0004.desc
        theme = emperor
    override_background = { reference = wilderness_mountains }

    left_portrait = {
        character = root
        animation = steward
    }
    option = {    
        name = eoe_roman_event.0004.a
        hidden_effect ={
            every_vassal_or_below = {
			    limit = {
				    capital_province = {
					    geographical_region = eoe_raetia_custom
				    }
				    is_ai = yes
			    }
			    random = {
				    chance = 60
                    set_culture = culture:rhaetio_roman
                    set_character_faith = root.faith
				    capital_county = {
					    set_county_culture = culture:rhaetio_roman
                        set_county_faith = root.faith
				    }
        	    }
		    }
        }
    }
    after = {
		eoe_clear_resettled_county_resistance_effect = {
			REGION = eoe_raetia_custom
			CULTURE = culture:rhaetio_roman
		}
        add_character_modifier = {
            modifier = eoe_forced_migration
            years = 10
        }
    }
}

eoe_roman_event.0015 = {
        type = character_event
        title = eoe_roman_event.0015.title
        desc = eoe_roman_event.0015.desc
        theme = faith
    override_background = { reference = temple }
    override_effect_2d = { reference = smoke }

    left_portrait = {
        character = root
        animation = thinking
    }
    option = {
        name = eoe_roman_event.0015.a
        if = {
            limit = {
                has_cp6_dlc_trigger = yes
            }
            title:b_vaticano.title_province = {
                remove_building = holy_site_st_peters_basilica_01
                add_special_building = circus_of_nero_01
                set_holding_type = city_holding
            }
        }
        else = {
            title:b_vaticano.title_province = {
            remove_building = holy_site_cathedral_01
            add_special_building = circus_of_nero_01
            set_holding_type = city_holding
            }
        }
    }
    option = {
        name = eoe_roman_event.0015.b
        if = {
            limit = {
                has_cp6_dlc_trigger = yes
            }
            title:b_vaticano.title_province = {
                remove_building = holy_site_st_peters_basilica_01
                add_special_building = temple_of_sol_invictus_01
                set_holding_type = church_holding
            }
        }
        else = {
            title:b_vaticano.title_province = {
            remove_building = holy_site_cathedral_01
            add_special_building = temple_of_sol_invictus_01
            set_holding_type = church_holding
            }
        }
    }
