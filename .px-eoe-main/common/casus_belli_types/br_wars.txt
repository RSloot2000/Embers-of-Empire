great_war_of_restoration_cb = {
	icon = migration_cb
	group = event

	full_occupation_by_defender_gives_victory = no
	attacker_capital_gives_war_score = no
	check_all_defenders_for_ticking_war_score = yes
	defender_ticking_warscore_delay = { months = 4 }
	defender_ticking_warscore = 0.125
	imprisonment_by_defender_give_war_score = no
	defender_score_from_occupation_scale = 50
	attacker_score_from_occupation_scale = 50
	defender_score_from_battles_scale = 125
	attacker_score_from_battles_scale = 125
	ticking_war_score_targets_entire_realm = yes

	occupation_participation_mult = 0.5
	siege_participation_mult = 1
	battle_participation_mult = 7.5

	allow_hostages = no

	allowed_against_character = {
		is_landed = yes
		government_has_flag = government_is_feudal
		highest_held_title_tier >= tier_empire
	}

	white_peace_possible = no

	cost = {
	}

	target_titles = claim
	target_title_tier = all

	on_declaration = {
		on_declared_war = yes
	}

	should_invalidate = {
		scope:defender = {
			OR = {
				is_independent_ruler = no
				NOR = {
					is_landed = yes
					is_landless_ruler = yes
				}
			}
			
		}
	}

	on_invalidated_desc = {
		desc = br_great_war_of_defiance_war_invalidated_message
	}
	
	on_invalidated = {
		scope:attacker = {
			random_character_war = {
				limit = {
					using_cb = great_war_of_restoration_cb
				}
				every_war_attacker = {
					add_to_list = anti_wr_attackers
				}
			}
		}
		every_in_list = {
			list = anti_wr_attackers
			add_prestige = medium_prestige_gain
			add_piety = medium_piety_gain
		}
		scope:defender = {
			add_prestige = medium_prestige_gain
			remove_decision_cooldown = restore_the_west
			br_cleanup_effect = yes
		}
	}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = br_war_of_defiance_wr_loses_defender
			}
			triggered_desc = {
				trigger = { scope:attacker = { is_local_player = yes } }
				desc = br_war_of_defiance_wr_loses_attacker
			}
			desc = br_war_of_defiance_wr_loses_other
		}
	}

	on_victory = {
		scope:attacker = {
			random_character_war = {
				limit = {
					using_cb = great_war_of_restoration_cb
				}
				save_scope_as = wr_war
				every_war_attacker = {
					add_to_list = anti_wr_attackers
				}
				every_war_defender = {
					add_to_list = anti_wr_defenders
				}
			}
		}

		add_legitimacy_attacker_victory_effect = yes

		scope:defender = {
			every_vassal = {
				custom = custom.every_defender_vassal
				limit = {
					highest_held_title_tier >= tier_county
				}
				create_title_and_vassal_change = {
					type = independency
					save_scope_as = change
					add_claim_on_loss = no
				}
				becomes_independent = { change = scope:change }
				resolve_title_and_vassal_change = scope:change
			}
			br_cleanup_effect = yes
			add_legitimacy = monumental_legitimacy_loss
			remove_short_term_gold = ninety_percent_gold_value
			add_prestige = negative_seventy_five_percent_prestige_value
			trigger_event = {
				id = br_augustus_of_wr.005
				days = 1
			}
		}

		custom_tooltip = wr_war_of_defiance_gold_tt
		custom_tooltip = accolade_wr_attackers_all_gain_glory_tt
		hidden_effect = {
			every_in_list = {
				list = anti_wr_attackers
				add_short_term_gold = medium_gold_value
				if = {
					limit = {
						has_dlc_feature = accolades
					}
					accolade_attacker_war_end_glory_gain_very_high_effect = yes
				}
			}
		}
		scope:attacker = {
			rightfully_imprison_character_less_verbose_effect = { 
				TARGET = scope:defender
				IMPRISONER = scope:attacker
			}
		}

		scope:attacker = {
			add_prestige_experience = {
				value = major_prestige_value
			}
		}

		scope:defender = {
			add_prestige_level = -1
		}

		scope:attacker = { show_pow_release_message_effect = yes }

		scope:defender = {
			create_character_memory = {
				type = wr_defeated_defiance_war
				participants = {
					opponent = scope:attacker
				}
			}
		}
		every_in_list = {
			list = anti_wr_defenders
			limit = {
				NOT = {
					this = scope:defender
				}
			}
			create_character_memory = {
				type = wr_defeated_defiance_war_defenders
				participants = {
					western_roman_emperor = scope:defender
				}
			}
		}
		every_in_list = {
			list = anti_wr_attackers
			create_character_memory = {
				type = wr_defeated_defiance_war_attackers
				participants = {
					western_roman_emperor = scope:defender
				}
			}
		}
		
		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = major_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 10
			LOSER_ALLY_FAME_SCALE = 1
		}

		laamp_as_mercenary_payout_tooltip_effect = yes

		add_truce_attacker_victory_effect = yes
	}

	on_white_peace_desc = {}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = war_of_defiance_wr_wins_defender
			}
			desc = war_of_defiance_wr_wins_other
		}
		
	}

	on_defeat = {
		scope:defender = {
			random_character_war = {
				limit = {
					using_cb = great_war_of_restoration_cb
				}
				every_war_attacker = {
					add_to_list = anti_wr_attackers
				}
				every_war_defender = {
					add_to_list = anti_wr_defenders
				}
			}
			trigger_event = {
				id = br_augustus_of_wr.004
				days = 1
			}
		}
		
		scope:defender = {
			custom_tooltip = will_become_augustus_of_wr_tt
			primary_title = {
				save_scope_as = defender_title
			}
			
			add_legitimacy = monumental_legitimacy_gain
			add_prestige = monumental_prestige_gain
			
			custom_tooltip = wr_great_war_of_defiance_attackers_gold_tt
			hidden_effect = {
				every_in_list = {
					list = anti_wr_attackers
					limit = {
						NOT = {
							this = scope:attacker
						}
					}
					pay_short_term_gold_reparations_effect = {
						GOLD_VALUE = 2
					}
				}
			}
		}
		add_truce_attacker_defeat_effect = yes

		scope:defender = {
			custom_tooltip = wr_delete_all_truces_tt
			hidden_effect = {
				every_truce_target = {
					scope:defender = { cancel_truce_one_way = prev }
				}
			}
		}

		if = {
			limit = {
				has_dlc_feature = accolades
			}
			custom_tooltip = accolade_defenders_all_gain_glory_tt
			hidden_effect = {
				scope:defender = {
					every_active_accolade = {
						custom = custom.every_active_accolade
						add_glory = massive_glory_gain
					}
				}
				every_in_list = {
					list = anti_wr_defenders
					limit = {
						NOT = {
							this = scope:defender
						}
					}
					every_active_accolade = {
						add_glory = major_glory_gain
					}
				}
			}
		}

		scope:attacker = {
			add_legitimacy = medium_legitimacy_loss
			pay_short_term_gold_reparations_effect = {
				GOLD_VALUE = 5
			}
		}

		scope:defender = {
			create_character_memory = {
				type = wr_won_defiance_war
				participants = {
					opponent = scope:attacker
				}
			}
		}

		every_in_list = {
			list = anti_wr_defenders
			limit = {
				NOT = {
					this = scope:defender
				}
			}
			create_character_memory = {
				type = wr_won_defiance_war_defenders
				participants = {
					western_roman_emperor = scope:defender
				}
			}
		}
		
		every_in_list = {
			list = anti_wr_attackers
			create_character_memory = {
				type = wr_won_defiance_war_attackers
				participants = {
					western_roman_emperor = scope:defender
				}
			}
		}

		modify_allies_of_participants_fame_values = {
			WINNER = scope:defender
			FAME_BASE = major_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 10
			LOSER_ALLY_FAME_SCALE = 1
		}

		laamp_as_mercenary_payout_tooltip_effect = yes
	}

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	check_defender_inheritance_validity = yes

	transfer_behavior = transfer

	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = "WR_GREAT_WAR_OF_DEFIANCE_NAME"
	my_war_name = "WR_GREAT_WAR_OF_DEFIANCE_NAME_MY"
	war_name_base = "WR_GREAT_WAR_OF_DEFIANCE_NAME_BASE"
	cb_name = "WR_GREAT_WAR_OF_DEFIANCE_NAME"
	interface_priority = 60

	should_show_war_goal_subview = yes

	attacker_wargoal_percentage = 0.5
	
	max_defender_score_from_occupation = 75
	max_attacker_score_from_occupation = 150

	max_ai_diplo_distance_to_title = 10000
}
