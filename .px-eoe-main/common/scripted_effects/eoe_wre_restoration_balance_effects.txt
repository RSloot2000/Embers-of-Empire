eoe_set_wre_restoration_war_active_effect = {
	if = {
		limit = { has_title = title:e_western_roman_empire }
		title:e_western_roman_empire = {
			set_variable = {
				name = eoe_wre_restoration_war_active
				value = yes
			}
		}
	}
	else_if = {
		limit = { has_title = title:h_western_roman_empire }
		title:h_western_roman_empire = {
			set_variable = {
				name = eoe_wre_restoration_war_active
				value = yes
			}
		}
	}
}

eoe_clear_wre_restoration_war_active_effect = {
	title:e_western_roman_empire = {
		remove_variable = eoe_wre_restoration_war_active
	}
	title:h_western_roman_empire = {
		remove_variable = eoe_wre_restoration_war_active
	}
}

eoe_apply_wre_diocesan_overextension_effect = {
	scope:attacker = {
		remove_character_modifier = eoe_wre_overextension_minor
		remove_character_modifier = eoe_wre_overextension_moderate
		remove_character_modifier = eoe_wre_overextension_major
		remove_character_modifier = eoe_wre_overextension_severe
	}

	if = {
		limit = { eoe_wre_restoration_targets_in_region_at_least = { REGION = $REGION$ COUNT = 31 } }
		scope:attacker = {
			add_character_modifier = { modifier = eoe_wre_overextension_severe years = 7 }
		}
	}
	else_if = {
		limit = { eoe_wre_restoration_targets_in_region_at_least = { REGION = $REGION$ COUNT = 16 } }
		scope:attacker = {
			add_character_modifier = { modifier = eoe_wre_overextension_major years = 5 }
		}
	}
	else_if = {
		limit = { eoe_wre_restoration_targets_in_region_at_least = { REGION = $REGION$ COUNT = 6 } }
		scope:attacker = {
			add_character_modifier = { modifier = eoe_wre_overextension_moderate years = 3 }
		}
	}
	else = {
		scope:attacker = {
			add_character_modifier = { modifier = eoe_wre_overextension_minor years = 2 }
		}
	}
}

eoe_set_wre_restoration_recovery_effect = {
	eoe_clear_wre_restoration_war_active_effect = yes
	add_character_modifier = {
		modifier = eoe_wre_imperial_recovery
		days = $DAYS$
	}
	if = {
		limit = { has_title = title:e_western_roman_empire }
		title:e_western_roman_empire = {
			set_variable = { name = eoe_wre_imperial_recovery value = yes days = $DAYS$ }
		}
	}
	else = {
		title:h_western_roman_empire = {
			set_variable = { name = eoe_wre_imperial_recovery value = yes days = $DAYS$ }
		}
	}
}

eoe_declare_wre_restoration_war_effect = {
	on_declared_war = yes
	scope:attacker = { eoe_set_wre_restoration_war_active_effect = yes }
	scope:war = {
		set_variable = { name = eoe_wre_campaign_threat value = 10 }
	}
	if = {
		limit = { eoe_wre_restoration_targets_in_region_at_least = { REGION = $REGION$ COUNT = 25 } }
		scope:war = { set_variable = { name = eoe_wre_campaign_threat value = 70 } }
	}
	else_if = {
		limit = { eoe_wre_restoration_targets_in_region_at_least = { REGION = $REGION$ COUNT = 16 } }
		scope:war = { set_variable = { name = eoe_wre_campaign_threat value = 50 } }
	}
	else_if = {
		limit = { eoe_wre_restoration_targets_in_region_at_least = { REGION = $REGION$ COUNT = 10 } }
		scope:war = { set_variable = { name = eoe_wre_campaign_threat value = 30 } }
	}
	else_if = {
		limit = { eoe_wre_restoration_targets_in_region_at_least = { REGION = $REGION$ COUNT = 6 } }
		scope:war = { set_variable = { name = eoe_wre_campaign_threat value = 20 } }
	}
	scope:defender = { eoe_coordinate_wre_restoration_coalition_effect = yes }
}

eoe_invalidate_wre_restoration_war_effect = {
	scope:attacker = { eoe_set_wre_restoration_recovery_effect = { DAYS = 365 } }
}

eoe_resolve_wre_diocesan_victory_effect = {
	eoe_apply_wre_diocesan_overextension_effect = { REGION = $REGION$ }

	every_in_list = {
		list = target_titles
		remove_from_list = target_titles
	}
	every_county_in_region = {
		region = $REGION$
		limit = {
			eoe_is_valid_wre_restoration_target = yes
			NOT = { is_in_list = target_titles }
		}
		set_variable = {
			name = eoe_wre_previous_holder
			value = holder
		}
		set_variable = { name = eoe_wre_pending_settlement value = yes }
		set_variable = { name = eoe_wre_pending_provincial_settlement value = yes }
		add_to_list = target_titles
	}

	scope:attacker = {
		show_pow_release_message_effect = yes
		eoe_set_wre_restoration_recovery_effect = { DAYS = 1825 }
	}
	add_legitimacy_attacker_victory_effect = yes
	add_influence_attacker_victory_effect = yes

	create_title_and_vassal_change = {
		type = conquest_claim
		save_scope_as = eoe_wre_restoration_change
		add_claim_on_loss = yes
	}
	setup_claim_cb = {
		titles = target_titles
		attacker = scope:attacker
		defender = scope:defender
		claimant = scope:attacker
		change = scope:eoe_wre_restoration_change
	}
	resolve_title_and_vassal_change = scope:eoe_wre_restoration_change

	modify_all_participants_fame_values = {
		WINNER = scope:attacker
		LOSER = scope:defender
		FAME_BASE = scope:cb_prestige_factor
		IS_RELIGIOUS_WAR = no
		WINNER_FAME_SCALE = 10
		LOSER_FAME_SCALE = -10
		WINNER_ALLY_FAME_SCALE = 10
		LOSER_ALLY_FAME_SCALE = 10
	}
	add_truce_attacker_victory_effect = yes
}

eoe_mobilize_wre_restoration_defenders_effect = {

		limit = {
			scope:war.var:eoe_wre_campaign_threat >= 70
		}
		spawn_army = {
			name = eoe_wre_major_defensive_mobilization
			location = capital_province
			war = scope:war
			uses_supply = yes
			inheritable = no
			levies = 2000
			men_at_arms = {
				type = light_footmen
				stacks = 7
			}
			men_at_arms = {
				type = bowmen
				stacks = 6
			}
			men_at_arms = {
				type = light_horsemen
				stacks = 3
			}
		}
	}
	else_if = {
		limit = {
			scope:war.var:eoe_wre_campaign_threat >= 30
		}

			name = eoe_wre_minor_defensive_mobilization
			location = capital_province
			war = scope:war
			uses_supply = yes
			inheritable = no
			levies = 1200

				type = light_footmen
				stacks = 4
			}
			men_at_arms = {

				stacks = 3
			}
			men_at_arms = {

				stacks = 1
			}
		}
	}
}

eoe_coordinate_wre_restoration_coalition_effect = {
	if = {
		limit = {
			scope:war.var:eoe_wre_campaign_threat >= 30
		}
		scope:war = {
			set_variable = {
				name = eoe_wre_coalition_military_responses
				value = 0
			}
			set_variable = {
				name = eoe_wre_coalition_direct_responses
				value = 0
			}
			set_variable = {
				name = eoe_wre_vassal_military_responses
				value = 0
			}
		}
		ordered_independent_ruler = {
			limit = {
				eoe_valid_wre_coalition_candidate = yes
			}
			order_by = eoe_wre_coalition_candidate_priority_value
			max = 5
			add_to_list = eoe_wre_coalition_candidates
		}
		if = {
			limit = {
				any_in_list = {
					list = eoe_wre_coalition_candidates
					always = yes
				}
			}
			scope:defender = {
				trigger_event = {
					id = eoe_wre_restoration_balance.001
					days = 24
				}
			}
			every_in_list = {
				list = eoe_wre_coalition_candidates
				trigger_event = {
					id = eoe_wre_restoration_coalition.001
					days = { 7 21 }
				}
			}
		}
		else = {
			eoe_mobilize_wre_restoration_defenders_effect = yes
		}
	}
}
