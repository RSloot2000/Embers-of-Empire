eoe_grant_wre_previous_holder_package_effect = {
	create_title_and_vassal_change = {
		type = granted
		save_scope_as = eoe_wre_settlement_change
		add_claim_on_loss = no
	}
	scope:eoe_wre_settlement_emperor = {
		every_realm_county = {
			limit = {
				holder = scope:eoe_wre_settlement_emperor
				has_variable = eoe_wre_pending_settlement
				de_jure_liege = scope:eoe_wre_package_duchy
				var:eoe_wre_previous_holder = root
			}
			change_title_holder = {
				holder = root
				change = scope:eoe_wre_settlement_change
			}
			remove_variable = eoe_wre_pending_settlement
			remove_variable = eoe_wre_restoration_target
			remove_variable = eoe_wre_previous_holder
		}
	}
	change_liege = {
		liege = scope:eoe_wre_settlement_emperor
		change = scope:eoe_wre_settlement_change
	}
	resolve_title_and_vassal_change = scope:eoe_wre_settlement_change
}

eoe_grant_wre_duchy_package_effect = {
	create_title_and_vassal_change = {
		type = granted
		save_scope_as = eoe_wre_settlement_change
		add_claim_on_loss = no
	}
	scope:eoe_wre_settlement_emperor = {
		ordered_realm_county = {
			limit = {
				holder = scope:eoe_wre_settlement_emperor
				has_variable = eoe_wre_pending_settlement
				de_jure_liege = scope:eoe_wre_package_duchy
			}
			order_by = development_level
			max = 3
			change_title_holder = {
				holder = root
				change = scope:eoe_wre_settlement_change
			}
			if = {
				limit = { scope:eoe_wre_settlement_emperor = { has_character_flag = eoe_wre_imperial_settlement_policy } }
				add_county_modifier = {
					modifier = eoe_wre_imposed_ruler_resistance
					years = 7
				}
			}
			remove_variable = eoe_wre_pending_settlement
			remove_variable = eoe_wre_restoration_target
			remove_variable = eoe_wre_previous_holder
			remove_variable = eoe_wre_package_refusals
		}
	}
	change_liege = {
		liege = scope:eoe_wre_settlement_emperor
		change = scope:eoe_wre_settlement_change
	}
	resolve_title_and_vassal_change = scope:eoe_wre_settlement_change
}

eoe_create_and_grant_wre_local_package_effect = {
	scope:eoe_wre_package_anchor.title_province.culture = { save_scope_as = eoe_wre_package_culture }
	scope:eoe_wre_package_anchor.title_province.faith = { save_scope_as = eoe_wre_package_faith }
	create_character = {
		age = 30
		gender_female_chance = 20
		culture = scope:eoe_wre_package_culture
		faith = scope:eoe_wre_package_faith
		dynasty = generate
		random_traits = yes
		location = scope:eoe_wre_package_anchor.title_province
		save_scope_as = eoe_wre_generated_settlement_ruler
	}
	scope:eoe_wre_generated_settlement_ruler = {
		eoe_grant_wre_duchy_package_effect = yes
	}
}

	scope:eoe_wre_settlement_emperor = {
		every_realm_county = {
			limit = {
				has_variable = eoe_wre_pending_provincial_settlement
				de_jure_liege = scope:eoe_wre_package_duchy

		}

}

eoe_grant_wre_provincial_governorship_effect = {
	create_title_and_vassal_change = {
		type = granted
		save_scope_as = eoe_wre_provincial_change
		add_claim_on_loss = no
	}
	scope:eoe_wre_package_duchy = {
		change_title_holder = {
			holder = root
			change = scope:eoe_wre_provincial_change
		}
	}
	change_liege = {
		liege = scope:eoe_wre_settlement_emperor
		change = scope:eoe_wre_provincial_change
	}
	scope:eoe_wre_settlement_emperor = {
		every_realm_county = {
			limit = {
				has_variable = eoe_wre_pending_provincial_settlement
				de_jure_liege = scope:eoe_wre_package_duchy
				holder != root
				holder.primary_title = { tier = tier_county }
			}
			holder = {
				change_liege = {
					liege = scope:eoe_wre_provincial_governor
					change = scope:eoe_wre_provincial_change
				}
			}
		}
	}
	resolve_title_and_vassal_change = scope:eoe_wre_provincial_change
	eoe_clear_wre_provincial_package_effect = yes
}
