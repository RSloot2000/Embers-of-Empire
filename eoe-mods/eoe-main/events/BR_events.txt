br_augustus_of_wr.002 = {

		animation = marshal
	}

		character = scope:primary_coalition_defender

	}

		hidden_effect = {

                    NOT = {
                        this = root
                    }
                    is_landed = yes
                    sub_realm_size >= 20
                    government_has_flag = government_is_feudal
                    is_independent_ruler = yes
                }
                save_scope_as = primary_coalition_defender
            }
        }
        else_if = {
            limit = {
                any_in_list = {
                    list = anti_wr_coalition_members
                    current_military_strength > 0
                    is_landed = yes
                    sub_realm_size >= 12
                    NOT = {
                        this = root
                    }
                    government_has_flag = government_is_feudal
                    is_independent_ruler = yes
                }
            }
            ordered_in_list = {
                list = anti_wr_coalition_members
                order_by = current_military_strength
                limit = {
                    is_landed = yes
                    NOT = {
                        this = root
                    }
                    sub_realm_size >= 12
                    government_has_flag = government_is_feudal
                    is_independent_ruler = yes
                }
                save_scope_as = primary_coalition_defender
            }

                list = anti_wr_coalition_members
                order_by = sub_realm_size

                    highest_held_title_tier >= tier_duchy

                        this = root

                        count >= 4

                        }
                    }
                }

                limit = {
                    is_ai = yes
                    NOT = {
                        government_has_flag = government_is_herder

                    exists = title:k_france.holder

br_augustus_of_wr.003 = {
	hidden = yes
	immediate = {
		scope:primary_coalition_defender = {
			if = {
				limit = {
					OR = {
						any_truce_holder = {
							this = root
						}
						any_truce_target = {
							this = root
						}
					}
				}
				cancel_truce_both_ways = root
			}
			if = {
				limit = {
					is_vassal_or_below_of = root
				}
				create_title_and_vassal_change = {
					type = independency
					save_scope_as = change
				}
				becomes_independent = { change = scope:change }
				resolve_title_and_vassal_change = scope:change
			}
			#BEGIN THE WAR!
			start_war = {
				cb = great_war_of_restoration_cb
				target = root
				target_title = root.primary_title
			}
			random_character_war = {
				limit = {
					using_cb = great_war_of_restoration_cb
				}
				every_in_list = {
					list = anti_wr_coalition_members
					limit = {
						NOR = {
							is_defender_in_war = prev
							is_attacker_in_war = prev
						}
					}
					prev = {
						add_attacker = prev
					}
				}
			}
		}
		every_in_list = {
			list = anti_wr_coalition_members
			every_character_war = {
				limit = {
					NOR = {
						using_cb = great_war_of_restoration_cb
						using_cb = crusading_claim_cb
						AND = {
							primary_defender = prev
							NOT = {
								using_cb = peasant_war
							}
						}
						primary_defender = {
							is_ai = no
						}
					}
				}
				if = {
					limit = {
						primary_attacker = prev
					}
					end_war = white_peace
				}
				else = {
					remove_participant = prev
				}
			}
		}
		random_character_war = {
			limit = {
				using_cb = great_war_of_restoration_cb
			}
			save_scope_as = wr_war
		}
		if = {
			limit = {
				OR = {
					has_relation_potential_rival = scope:primary_coalition_defender
					scope:primary_coalition_defender = {
						opinion = {
							target = root
							value < -40
						}
					}
				}
			}
			set_relation_rival = {
				target = scope:primary_coalition_defender
				reason = rival_attempt_revive_wr
			}
		}
	}
}
