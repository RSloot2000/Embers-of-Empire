namespace = eoe_wre_restoration_coalition

eoe_wre_restoration_coalition.001 = {
	type = character_event
	title = eoe_wre_restoration_coalition.001.t
	desc = {
		desc = eoe_wre_restoration_coalition.001.desc
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:defender.capital_county.title_province = {
						geographical_region = gallia_diocese_region
					}
				}
				desc = eoe_wre_restoration_coalition.001.desc.francia
			}
			triggered_desc = {
				trigger = {
					scope:defender.capital_county.title_province = {
						geographical_region = hispania_diocese_region
					}
				}
				desc = eoe_wre_restoration_coalition.001.desc.iberia
			}
			triggered_desc = {
				trigger = {
					scope:defender.capital_county.title_province = {
						geographical_region = africae_diocese_region
					}
				}
				desc = eoe_wre_restoration_coalition.001.desc.africa
			}
			desc = eoe_wre_restoration_coalition.001.desc.generic
		}
	}
	theme = war
	override_background = { reference = army_camp }
	left_portrait = {
		character = root
		animation = thinking
	}
	right_portrait = {
		character = scope:defender
		animation = marshal
	}
	lower_center_portrait = {
		character = scope:attacker
		animation = anger
	}

	trigger = {
		exists = scope:war
		scope:war = {
			eoe_is_wre_restoration_war = yes
		}
		NOT = { is_attacker_in_war = scope:war }
		NOT = { is_defender_in_war = scope:war }
	}

	immediate = {
		add_character_flag = {
			flag = eoe_wre_coalition_invitation_cooldown
			years = 5
		}
	}

	option = {
		name = eoe_wre_restoration_coalition.001.a
		trigger = {
			OR = {
				scope:war.var:eoe_wre_campaign_threat >= 50
				AND = {
					scope:war.var:eoe_wre_campaign_threat >= 30
					scope:war = {
						war_defender_total_current_strength_value < war_attacker_total_current_strength_halved_value
					}
				}
			}
			scope:war = {
				var:eoe_wre_coalition_military_responses < 3
				var:eoe_wre_coalition_direct_responses < 1
				war_defender_total_current_strength_value < war_attacker_total_current_strength_value
			}
		}
		scope:war = {
			set_variable = {
				name = eoe_wre_coalition_formed
				value = yes
			}
			add_defender = root
			change_variable = {
				name = eoe_wre_coalition_military_responses
				add = 1
			}
			change_variable = {
				name = eoe_wre_coalition_direct_responses
				add = 1
			}
			if = {
				limit = {
					NOT = { has_variable = eoe_wre_coalition_attacker_notified }
				}
				set_variable = {
					name = eoe_wre_coalition_attacker_notified
					value = yes
				}
				scope:attacker = {
					trigger_event = eoe_wre_restoration_coalition.002
				}
			}
		}
		add_prestige = -100
		ai_chance = {
			base = 25
			modifier = {
				add = 35
				is_allied_to = scope:defender
			}
			modifier = {
				add = 25
				exists = dynasty
				exists = scope:defender.dynasty
				dynasty = scope:defender.dynasty
			}
			modifier = {
				add = 25
				scope:war.var:eoe_wre_campaign_threat >= 70
			}
			modifier = {
				add = 40
				scope:war = {
					war_defender_total_current_strength_value < war_attacker_total_current_strength_halved_value
				}
			}
			modifier = {
				add = -40
				has_relation_rival = scope:defender
			}
		}
	}

	option = {
		name = eoe_wre_restoration_coalition.001.b
		trigger = { gold >= 100 }
		remove_short_term_gold = 100
		scope:defender = { add_gold = 100 }
		ai_chance = { base = 30 }
	}

	option = {
		name = eoe_wre_restoration_coalition.001.c
		trigger = {
			scope:war = {
				var:eoe_wre_coalition_military_responses < 3
				war_defender_total_current_strength_value < war_attacker_total_current_strength_value
			}
		}
		scope:defender = {
			spawn_army = {
				name = eoe_wre_coalition_volunteers
				location = capital_province
				war = scope:war
				uses_supply = yes
				inheritable = no
				levies = 500
				men_at_arms = {
					type = light_footmen
					stacks = 2
				}
				men_at_arms = {
					type = bowmen
					stacks = 2
				}
			}
		}
		scope:war = {
			change_variable = {
				name = eoe_wre_coalition_military_responses
				add = 1
			}
		}
		add_prestige = -50
		ai_chance = { base = 35 }
	}

	option = {
		name = eoe_wre_restoration_coalition.001.d
		add_prestige = -50
		scope:defender = { add_prestige = 150 }
		ai_chance = { base = 20 }
	}

	option = {
		name = eoe_wre_restoration_coalition.001.e
		ai_chance = { base = 20 }
	}

	option = {
		name = eoe_wre_restoration_coalition.001.f
		add_prestige = 100
		ai_chance = {
			base = 10
			modifier = {
				add = 40
				has_relation_rival = scope:defender
			}
		}
	}
}
