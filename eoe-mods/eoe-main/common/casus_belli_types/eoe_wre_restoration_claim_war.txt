eoe_wre_restoration_claim_cb = {
	icon = claim_cb
	group = claim
	combine_into_one = yes
	should_show_war_goal_subview = no
	mutually_exclusive_titles = { always = yes }

	allowed_for_character = {
		always = no
		OR = {
			has_title = title:e_western_roman_empire
			has_title = title:h_western_roman_empire
		}
		custom_description = {
			text = eoe_wre_imperial_recovery_blocks_cb
			NOR = {
				title:e_western_roman_empire = {
					has_variable = eoe_wre_imperial_recovery
				}
				title:h_western_roman_empire = {
					has_variable = eoe_wre_imperial_recovery
				}
			}
		}
		custom_description = {
			text = eoe_wre_restoration_war_active_blocks_cb
			NOR = {
				title:e_western_roman_empire = {
					has_variable = eoe_wre_restoration_war_active
				}
				title:h_western_roman_empire = {
					has_variable = eoe_wre_restoration_war_active
				}
			}
		}
	}

	allowed_against_character = {
		is_landed = yes
		is_independent_ruler = yes
		scope:attacker.top_liege != scope:defender.top_liege
		eoe_wre_restoration_targets_at_least = {
			COUNT = 1
		}
	}

	target_titles = all
	target_title_tier = all

	valid_to_start = {
		scope:target = {
			this = scope:defender.primary_title
		}
	}

	war_name = "EOE_WRE_RESTORATION_CLAIM_WAR_NAME"
	my_war_name = "EOE_WRE_RESTORATION_CLAIM_WAR_NAME_MY"
	war_name_base = "EOE_WRE_RESTORATION_CLAIM_WAR_NAME_BASE"
	cb_name = "eoe_wre_restoration_claim_cb"
	interface_priority = 100

	ticking_war_score_targets_entire_realm = yes

	cost = {
		prestige = {
			value = eoe_wre_restoration_prestige_cost_value
			multiply = common_cb_prestige_cost_multiplier
		}
		gold = {
			value = eoe_wre_restoration_gold_cost_value
		}
	}

	on_declaration = {
		eoe_declare_wre_restoration_war_effect = { REGION = custom_western_roman_empire }
	}

	should_invalidate = {
		OR = {
			scope:defender = {
				OR = {
					is_independent_ruler = no
					is_landed = no
				}
			}
			NOT = {
				eoe_wre_restoration_targets_at_least = {
					COUNT = 1
				}
			}
		}
	}

	on_invalidated_desc = {
		desc = eoe_wre_restoration_claim_cb_invalidated_desc
	}

	on_invalidated = {
		scope:attacker = {
			eoe_clear_wre_restoration_war_active_effect = yes
			add_character_modifier = {
				modifier = eoe_wre_imperial_recovery
				years = 1
			}
			if = {
				limit = { has_title = title:e_western_roman_empire }
				title:e_western_roman_empire = {
					set_variable = {
						name = eoe_wre_imperial_recovery
						value = yes
						days = 365
					}
				}
			}
			else = {
				title:h_western_roman_empire = {
					set_variable = {
						name = eoe_wre_imperial_recovery
						value = yes
						days = 365
					}
				}
			}
		}
	}

	on_victory_desc = {
		desc = eoe_wre_restoration_claim_cb_victory_desc
	}

	on_victory = {
		scope:attacker = {
			eoe_clear_wre_restoration_war_active_effect = yes
			eoe_apply_wre_restoration_overextension_effect = yes
		}

		every_in_list = {
			list = target_titles
			remove_from_list = target_titles
		}

		every_county_in_region = {
			region = custom_western_roman_empire
			limit = {
				eoe_is_valid_wre_restoration_target = yes
				NOT = { is_in_list = target_titles }
			}
			add_to_list = target_titles
		}

		scope:attacker = {
			show_pow_release_message_effect = yes
			add_character_modifier = {
				modifier = eoe_wre_imperial_recovery
				years = 5
			}
			if = {
				limit = { has_title = title:e_western_roman_empire }
				title:e_western_roman_empire = {
					set_variable = {
						name = eoe_wre_imperial_recovery
						value = yes
						days = 1825
					}
				}
			}
			else = {
				title:h_western_roman_empire = {
					set_variable = {
						name = eoe_wre_imperial_recovery
						value = yes
						days = 1825
					}
				}
			}
		}
		add_legitimacy_attacker_victory_effect = yes
		add_influence_attacker_victory_effect = yes

		create_title_and_vassal_change = {
			type = conquest_claim
			save_scope_as = eoe_wre_restoration_change
			add_claim_on_loss = yes
		}
		setup_claim_cb = {
			titles = target_titles
			attacker = scope:attacker
			defender = scope:defender
			claimant = scope:attacker
			change = scope:eoe_wre_restoration_change
		}
		resolve_title_and_vassal_change = scope:eoe_wre_restoration_change

		modify_all_participants_fame_values = {
			WINNER = scope:attacker
			LOSER = scope:defender
			FAME_BASE = scope:cb_prestige_factor
			IS_RELIGIOUS_WAR = no
			WINNER_FAME_SCALE = 10
			LOSER_FAME_SCALE = -10
			WINNER_ALLY_FAME_SCALE = 10
			LOSER_ALLY_FAME_SCALE = 10
		}
		add_truce_attacker_victory_effect = yes
	}

	on_white_peace_desc = {
		desc = eoe_wre_restoration_claim_cb_white_peace_desc
	}

	on_white_peace = {
		scope:attacker = {
			eoe_clear_wre_restoration_war_active_effect = yes
			show_pow_release_message_effect = yes
			add_character_modifier = {
				modifier = eoe_wre_imperial_recovery
				years = 3
			}
			if = {
				limit = { has_title = title:e_western_roman_empire }
				title:e_western_roman_empire = {
					set_variable = {
						name = eoe_wre_imperial_recovery
						value = yes
						days = 1095
					}
				}
			}
			else = {
				title:h_western_roman_empire = {
					set_variable = {
						name = eoe_wre_imperial_recovery
						value = yes
						days = 1095
					}
				}
			}
		}
		add_truce_white_peace_effect = yes
	}

	on_defeat_desc = {
		desc = eoe_wre_restoration_claim_cb_defeat_desc
	}

	on_defeat = {
		scope:attacker = {
			eoe_clear_wre_restoration_war_active_effect = yes
			show_pow_release_message_effect = yes
			add_character_modifier = {
				modifier = eoe_wre_imperial_recovery
				years = 8
			}
			if = {
				limit = { has_title = title:e_western_roman_empire }
				title:e_western_roman_empire = {
					set_variable = {
						name = eoe_wre_imperial_recovery
						value = yes
						days = 2920
					}
				}
			}
			else = {
				title:h_western_roman_empire = {
					set_variable = {
						name = eoe_wre_imperial_recovery
						value = yes
						days = 2920
					}
				}
			}
			pay_short_term_gold_reparations_effect = {
				GOLD_VALUE = 3
			}
		}
		add_legitimacy_attacker_defeat_effect = yes
		add_influence_attacker_defeat_effect = yes
		add_truce_attacker_defeat_effect = yes
	}
}
