eoe_wre_restoration_gallia_cb = {
	icon = claim_cb
	group = claim
	combine_into_one = yes
	should_show_war_goal_subview = no
	mutually_exclusive_titles = { always = yes }

	allowed_for_character = {
		OR = {
			has_title = title:e_western_roman_empire
			has_title = title:h_western_roman_empire
		}
		custom_description = {
			text = eoe_wre_imperial_recovery_blocks_cb
			NOR = {
				title:e_western_roman_empire = { has_variable = eoe_wre_imperial_recovery }
				title:h_western_roman_empire = { has_variable = eoe_wre_imperial_recovery }
			}
		}
		custom_description = {
			text = eoe_wre_restoration_war_active_blocks_cb
			NOR = {
				title:e_western_roman_empire = { has_variable = eoe_wre_restoration_war_active }
				title:h_western_roman_empire = { has_variable = eoe_wre_restoration_war_active }
			}
		}
	}

	allowed_against_character = {
		is_landed = yes
		is_independent_ruler = yes
		scope:attacker.top_liege != scope:defender.top_liege
		eoe_wre_restoration_targets_in_region_at_least = {
			REGION = gallia_diocese_region
			COUNT = 1
		}
	}

	target_titles = all
	target_title_tier = all
	valid_to_start = { scope:target = { this = scope:defender.primary_title } }

	war_name = "EOE_WRE_RESTORATION_GALLIA_WAR_NAME"
	my_war_name = "EOE_WRE_RESTORATION_GALLIA_WAR_NAME_MY"
	war_name_base = "EOE_WRE_RESTORATION_GALLIA_WAR_NAME_BASE"
	cb_name = "eoe_wre_restoration_gallia_cb"
	interface_priority = 110
	ticking_war_score_targets_entire_realm = yes

	cost = {
		prestige = {
			value = eoe_wre_restoration_gallia_prestige_cost_value
			multiply = common_cb_prestige_cost_multiplier
		}
		gold = { value = eoe_wre_restoration_gallia_gold_cost_value }
	}

	on_declaration = {
		eoe_declare_wre_restoration_war_effect = { REGION = gallia_diocese_region }
	}

	should_invalidate = {
		OR = {
			scope:defender = { OR = { is_independent_ruler = no is_landed = no } }
			NOT = { eoe_wre_restoration_targets_in_region_at_least = { REGION = gallia_diocese_region COUNT = 1 } }
		}
	}

	on_invalidated_desc = { desc = eoe_wre_restoration_claim_cb_invalidated_desc }
	on_invalidated = { eoe_invalidate_wre_restoration_war_effect = yes }

	on_victory_desc = { desc = eoe_wre_restoration_gallia_cb_victory_desc }
	on_victory = { eoe_resolve_wre_diocesan_victory_effect = { REGION = gallia_diocese_region } }

	on_white_peace_desc = { desc = eoe_wre_restoration_claim_cb_white_peace_desc }
	on_white_peace = { eoe_white_peace_wre_restoration_war_effect = yes }

	on_defeat_desc = { desc = eoe_wre_restoration_claim_cb_defeat_desc }
	on_defeat = { eoe_defeat_wre_restoration_war_effect = yes }
